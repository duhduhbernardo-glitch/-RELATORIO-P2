// --- GERAÇÃO DE PDF E REDIRECIONAMENTO WHATSAPP --- function imprimirEFinalizar() { // Oculta botões para não saírem no PDF const elementosParaOcultar = document.querySelectorAll('.nav-tabs, .btn-add, .btn-action, #palletForm, #defectForm, .no-print'); elementosParaOcultar.forEach(el => el.style.display = 'none'); const turno = document.getElementById('selectTurno').value; const dataHoje = new Date().toLocaleDateString('pt-BR').replace(/\//g, '-'); const nomeArquivo = `Relatorio_Turno_${turno}_${dataHoje}.pdf`; const opcoes = { margin: 10, filename: nomeArquivo, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2 }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' } }; // 1. Gera e baixa o PDF instantaneamente html2pdf().set(opcoes).from(document.body).save().then(() => { // Restaura a exibição dos botões na tela elementosParaOcultar.forEach(el => el.style.display = ''); setTimeout(() => { // NOTIFICAÇÃO 1: Confirmação de Download do PDF alert("📄 PDF baixado com sucesso na sua pasta 'Downloads'!"); setTimeout(() => { // NOTIFICAÇÃO 2: Aviso de redirecionamento para o WhatsApp alert("📲 Clique em OK para abrir o WhatsApp e anexar o arquivo PDF baixado."); // Abre o WhatsApp abrirWhatsAppSimples(); // NOTIFICAÇÃO 3: Pergunta se deseja zerar para o próximo turno setTimeout(() => { if (confirm("Deseja ZERAR os dados do sistema agora para dar início ao próximo turno?")) { zerarTurno(); } }, 800); }, 400); }, 300); }); }
